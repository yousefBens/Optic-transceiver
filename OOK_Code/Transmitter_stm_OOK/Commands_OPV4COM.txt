    };

    aliases {
        user-gpio  = &led1;

        my-spi0-01 = &spidev00;
        my-spi0-02 = &spidev01;

        my-spi1-01 = &spidev10;
        my-spi1-02 = &spidev11;
    };

/* UART0 pour la console */
&uart0 {
    compatible = "nordic,nrf-uarte";
    status = "okay";
    current-speed = <115200>;
    pinctrl-0 = <&uart0_default>;
    pinctrl-1 = <&uart0_sleep>;
    pinctrl-names = "default", "sleep";
};

/* ================= SPI0 : 2 CS ================= */

&spi0 {
    status = "okay";

    /* Deux lignes de Chip Select pour spi0 */
    cs-gpios = <&gpio0 4 GPIO_ACTIVE_LOW>,   /* index 0 */
               <&gpio0 28 GPIO_ACTIVE_LOW>;   /* index 1 */

    pinctrl-0 = <&spi0_default>;
    pinctrl-1 = <&spi0_sleep>;
    pinctrl-names = "default", "sleep";

    /* Device sur CS0 (index 0 de cs-gpios) */
    spidev00: spidev@0 {
        compatible = "zephyr,spi-device";
        reg = <0>;                    /* -> utilise cs-gpios[0] (P0.03) */
        spi-max-frequency = <4000000>;
        /* tu peux ajouter mode, etc. */
    };

    /* Device sur CS1 (index 1 de cs-gpios) */
    spidev01: spidev@1 {
        compatible = "zephyr,spi-device";
        reg = <1>;                    /* -> utilise cs-gpios[1] (P0.04) */
        spi-max-frequency = <4000000>;
    };
};

/* ================= SPI1 : 2 CS ================= */

&spi1 {
    status = "okay";

    /* Deux lignes de Chip Select pour spi1 */
    cs-gpios = <&gpio0 15 GPIO_ACTIVE_LOW>,  /* index 0 */
               <&gpio0 16 GPIO_ACTIVE_LOW>;  /* index 1 */

    pinctrl-0 = <&spi1_default>;
    pinctrl-1 = <&spi1_sleep>;
    pinctrl-names = "default", "sleep";

    /* Device sur CS0 de spi1 */
    spidev10: spidev@0 {
        compatible = "zephyr,spi-device";
        reg = <0>;                    /* -> cs-gpios[0] (P0.15) */
        spi-max-frequency = <4000000>;
    };

    /* Device sur CS1 de spi1 */
    spidev11: spidev@1 {
        compatible = "zephyr,spi-device";
        reg = <1>;                    /* -> cs-gpios[1] (P0.16) */
        spi-max-frequency = <4000000>;
    };
};
